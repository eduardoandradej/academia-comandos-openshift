## Desafio sem pista — uma solução possível

```bash
audita_chart() {
  local out; out=$(helm template auditoria "$1" -n ac14-helm 2>/dev/null) || { echo "não renderiza"; return 1; }
  echo "tipos: $(grep -E '^kind:' <<<"$out" | awk '{print $2}' | sort | uniq -c | tr -s ' ' | paste -sd,)"
  local cl; cl=$(grep -E '^kind: (ClusterRole|ClusterRoleBinding|CustomResourceDefinition|SecurityContextConstraints|Namespace)$' <<<"$out" | sort -u | paste -sd,)
  echo "escopo de cluster: ${cl:-nenhum}"
  echo "imagens sem digest: $(grep -E '^\s+image:' <<<"$out" | grep -vc '@sha256:')"
  echo "requests.cpu: $(python3 -c 'import sys,yaml
t=0
for d in yaml.safe_load_all(sys.stdin):
  for c in ((d or {}).get("spec",{}).get("template",{}).get("spec",{}).get("containers",[]) if d else []):
    v=str(c.get("resources",{}).get("requests",{}).get("cpu","0")); t+= int(v[:-1]) if v.endswith("m") else int(float(v)*1000)
print(f"{t}m")' <<<"$out")"
  [ -z "$cl" ] && echo "instalável na Academia: verificar quota e imagens" || echo "instalável na Academia: não (pede escopo de cluster)"
}
audita_chart /tmp/ac14/terceiros/*
```

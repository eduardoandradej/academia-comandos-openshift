## Desafio sem pista — uma solução possível

```bash
audita() {
  local n=$1 r=~/relatorios/auditoria-$1.md
  regra() { if eval "$2" >/dev/null 2>&1; then echo "- conforme: $1"; else echo "- **não conforme**: $1 — prova: \`${2:0:120}\`"; fi; }
  {
    echo "# Auditoria $n — $(date +%F)"; echo
    regra "imagens por digest nos pods" "! oc get pods -n $n -o jsonpath='{range .items[*].spec.containers[*]}{.image}{\"\n\"}{end}' | grep -v '@sha256:' | grep -q ."
    regra "requests/limits/probes" "! oc get deploy -n $n -o json | jq -e '[.items[].spec.template.spec.containers[] | select(.resources.limits == null or .readinessProbe == null or .livenessProbe == null)] | length > 0'"
    regra "sem Secret em env" "! oc get deploy -n $n -o json | grep -qE 'secretKeyRef|secretRef'"
    regra "Routes com TLS e Redirect" "! oc get routes -n $n -o json | jq -e '[.items[] | select(.spec.tls.insecureEdgeTerminationPolicy != \"Redirect\")] | length > 0'"
    regra "sem BestEffort e reinícios <= 3" "! oc get pods -n $n -o json | jq -e '[.items[] | select(.status.qosClass==\"BestEffort\" or ([.status.containerStatuses[]?.restartCount] | max // 0) > 3)] | length > 0'"
    regra "NetworkPolicy de entrada" "oc get networkpolicy -n $n -o json | jq -e '[.items[] | select(.spec.policyTypes | index(\"Ingress\"))] | length > 0'"
    regra "rótulo da Academia" "! oc get deploy,svc,route,cm -n $n -o json | jq -e '[.items[] | select(.metadata.labels[\"academy.openshift.dev/managed-by\"] != \"academia\")] | length > 0'"
  } > "$r"; cat "$r"
}
audita ac20-loja
```

## Desafio sem pista — uma solução possível

```bash
parecer_cp() {
  local p=()
  oc get --raw /readyz >/dev/null 2>&1 || p+=("API não pronta")
  oc get co etcd -o jsonpath='{.status.conditions[?(@.type=="Available")].status}' | grep -q True || p+=("etcd indisponível")
  [ "$(oc get csr --no-headers 2>/dev/null | grep -c Pending)" -eq 0 ] || p+=("CSRs pendentes")
  [ -z "$(oc get co -o json | jq -r '.items[] | select(any(.status.conditions[]; .type=="Degraded" and .status=="True")) | .metadata.name')" ] || p+=("operadores degradados")
  local prox; prox=$(oc get secrets -n openshift-kube-apiserver-operator -o json | jq -r '[.items[].metadata.annotations["auth.openshift.io/certificate-not-after"] // empty] | sort | first')
  [ -n "$prox" ] && [ $(( ($(date -d "$prox" +%s) - $(date +%s)) / 86400 )) -lt 7 ] && p+=("certificado expira em < 7 dias")
  [ ${#p[@]} -eq 0 ] && echo "saudável" || printf 'investigar: %s\n' "${p[@]}"
}
parecer_cp
```

## Diagnóstico controlado — explicação

CSRs pendentes após longos desligamentos são normais em SNO; a aprovação segue o procedimento documentado do HomeLab.

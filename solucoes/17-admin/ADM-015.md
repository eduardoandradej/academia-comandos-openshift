oc describe node | sed -n '/Allocated resources:/,/Events:/p'; oc adm top node                 # 1

oc apply -f manifests/admin/clusterresourcequota-ac17.yaml; oc label ns <p> academy.openshift.dev/equipe=ac17   # 7
oc get node -o jsonpath='{.items[0].spec.taints}'; oc get scheduler cluster -o yaml            # 8

# 10: varredura (ADM-010); 11: parecer_cp (ADM-011)

# 14: por_que_nao_sobe (ADM-014); 15: limpeza acima

## Desafio sem pista — uma solução possível

```bash
sincroniza() {
  local nome=$1 arq=$2 obj
  obj=$(oc create configmap "$nome" --from-env-file="$arq" --dry-run=client -o yaml \
        | oc label --local -f - academy.openshift.dev/managed-by=academia -o yaml)
  if printf '%s\n' "$obj" | oc diff -n ac08-config -f -; then echo "nada a mudar"; return 0; fi
  read -r -p "aplicar? (s/N) " r; [ "$r" = s ] && printf '%s\n' "$obj" | oc apply -n ac08-config -f -
}
sincroniza ac08-env /tmp/ac08/app.env
```

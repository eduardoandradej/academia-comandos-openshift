## Desafio sem pista — uma solução possível

```bash
pode() {
  local sa=$1; shift
  while [ $# -ge 2 ]; do
    printf '%-10s %-20s %s\n' "$1" "$2" "$(oc auth can-i "$1" "$2" -n ac15-rbac --as=system:serviceaccount:ac15-rbac:"$sa")"
    shift 2
  done
}
pode ac15-leitor get pods get pods/log list configmaps create deployments
```

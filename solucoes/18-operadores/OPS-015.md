## Desafio sem pista — uma solução possível

```bash
mkdir -p ~/relatorios; R=~/relatorios/saude-$(date +%F).md
{
  echo "# Saúde do HomeLab — $(date +%F)"
  echo; echo "- versão: $(oc get clusterversion version -o jsonpath='{.status.desired.version}')"
  echo "- operadores fora do ideal: $(oc get co --no-headers | awk '$3!="True" || $4!="False" || $5!="False" {printf "%s ", $1}')"
  echo "- MCP master: $(oc get mcp master -o jsonpath='Updated={.status.conditions[?(@.type=="Updated")].status} Degraded={.status.conditions[?(@.type=="Degraded")].status}')"
  echo "- nó: $(oc adm top node --no-headers 2>/dev/null | awk '{print "CPU " $3 ", memória " $5}')"
  echo "- alertas: $(declare -F alertas >/dev/null && alertas | wc -l || echo 'rode a função do OPS-013')"
  echo "- atualização: $(declare -F pronto_para_atualizar >/dev/null && pronto_para_atualizar | head -1 || echo 'rode a função do OPS-011')"
} > "$R"; cat "$R"
```

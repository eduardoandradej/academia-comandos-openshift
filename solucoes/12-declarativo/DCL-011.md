## Desafio sem pista — uma solução possível

```bash
implanta() {
  local d=$1 ns=${2:-ac12-declarativo} dep
  oc apply --dry-run=server -R -f "$d" -n "$ns" >/dev/null || { echo "validação falhou"; return 1; }
  oc apply -R -f "$d" -n "$ns" || return 1
  for dep in $(oc create --dry-run=client -R -f "$d" -o json | jq -r '(.items // [.])[] | select(.kind=="Deployment") | .metadata.name'); do
    oc wait --for=condition=Available "deployment/$dep" -n "$ns" --timeout=120s || { echo "$dep não ficou disponível"; return 1; }
  done
  echo "implantado"
}
implanta /tmp/ac12/dir
```
